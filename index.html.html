    <script>
        // --- Three.js 3D Floating Hearts ---
        const initThreeHearts = () => {
            const container = document.getElementById('webgl-container');
            const scene = new THREE.Scene();
            const camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
            const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true });
            
            renderer.setSize(window.innerWidth, window.innerHeight);
            renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
            container.appendChild(renderer.domElement);

            // Lighting
            const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);
            scene.add(ambientLight);
            const dirLight = new THREE.DirectionalLight(0xffb6c1, 1.2);
            dirLight.position.set(5, 5, 5);
            scene.add(dirLight);

            // Create Heart Geometry
            const shape = new THREE.Shape();
            const x = 0, y = 0;
            shape.moveTo(x + 5, y + 5);
            shape.bezierCurveTo(x + 5, y + 5, x + 4, y, x, y);
            shape.bezierCurveTo(x - 6, y, x - 6, y + 7, x - 6, y + 7);
            shape.bezierCurveTo(x - 6, y + 11, x - 3, y + 15.4, x + 5, y + 19);
            shape.bezierCurveTo(x + 12, y + 15.4, x + 16, y + 11, x + 16, y + 7);
            shape.bezierCurveTo(x + 16, y + 7, x + 16, y, x + 10, y);
            shape.bezierCurveTo(x + 7, y, x + 5, y + 5, x + 5, y + 5);

            const extrudeSettings = { depth: 2, bevelEnabled: true, bevelSegments: 3, steps: 2, bevelSize: 1, bevelThickness: 1 };
            const geometry = new THREE.ExtrudeGeometry(shape, extrudeSettings);
            
            // Center the geometry
            geometry.computeBoundingBox();
            const centerOffset = -0.5 * (geometry.boundingBox.max.x - geometry.boundingBox.min.x);
            const centerYOffset = -0.5 * (geometry.boundingBox.max.y - geometry.boundingBox.min.y);
            geometry.translate(centerOffset, centerYOffset, 0);
            geometry.scale(0.04, 0.04, 0.04);
            geometry.rotateX(Math.PI); // Flip heart upright

            const material = new THREE.MeshPhysicalMaterial({
                color: 0xff3366,
                metalness: 0.1,
                roughness: 0.2,
                clearcoat: 1.0,
                clearcoatRoughness: 0.1,
                transparent: true,
                opacity: 0.85
            });

            const hearts = [];
            for (let i = 0; i < 24; i++) {
                const mesh = new THREE.Mesh(geometry, material.clone());
                mesh.position.set(
                    (Math.random() - 0.5) * 20,
                    (Math.random() - 0.5) * 15,
                    (Math.random() - 0.5) * 10 - 5
                );
                mesh.rotation.set(
                    Math.random() * Math.PI,
                    Math.random() * Math.PI,
                    Math.random() * Math.PI
                );
                
                const scale = 0.5 + Math.random() * 0.7;
                mesh.scale.set(scale, scale, scale);

                scene.add(mesh);
                hearts.push({
                    mesh,
                    speedX: (Math.random() - 0.5) * 0.015,
                    speedY: 0.005 + Math.random() * 0.015,
                    speedRotX: (Math.random() - 0.5) * 0.02,
                    speedRotY: (Math.random() - 0.5) * 0.02,
                    yOffset: Math.random() * Math.PI * 2
                });
            }

            camera.position.z = 10;
            let isFinalAnimation = false;

            const animate = () => {
                requestAnimationFrame(animate);
                const time = Date.now() * 0.001;

                hearts.forEach((heart) => {
                    if (!isFinalAnimation) {
                        // Gentle background floating
                        heart.mesh.position.y += Math.sin(time + heart.yOffset) * 0.003 + heart.speedY;
                        heart.mesh.position.x += Math.cos(time + heart.yOffset) * 0.002 + heart.speedX;
                        heart.mesh.rotation.x += heart.speedRotX;
                        heart.mesh.rotation.y += heart.speedRotY;

                        // Wrap around top
                        if (heart.mesh.position.y > 10) {
                            heart.mesh.position.y = -10;
                            heart.mesh.position.x = (Math.random() - 0.5) * 20;
                        }
                    } else {
                        // Swirl outwards and up for finale
                        heart.mesh.position.y += heart.speedY * 6;
                        heart.mesh.position.x += (heart.mesh.position.x > 0 ? 1 : -1) * 0.05 + Math.sin(time * 5 + heart.yOffset) * 0.05;
                        heart.mesh.position.z -= 0.03;
                        heart.mesh.rotation.x += heart.speedRotX * 4;
                        heart.mesh.rotation.y += heart.speedRotY * 4;
                        heart.mesh.material.opacity -= 0.01;
                        if(heart.mesh.material.opacity < 0) heart.mesh.material.opacity = 0;
                    }
                });

                renderer.render(scene, camera);
            };
            
            animate();

            window.addEventListener('resize', () => {
                camera.aspect = window.innerWidth / window.innerHeight;
                camera.updateProjectionMatrix();
                renderer.setSize(window.innerWidth, window.innerHeight);
            });

            // Expose function for the finale
            window.triggerHeartsFinale = () => {
                isFinalAnimation = true;
            };
        };
        
        initThreeHearts();

        // --- Step Navigation with GSAP ---
        let currentStep = 1;
        const totalSteps = 5;

        function updateProgress() {
            const progress = ((currentStep - 1) / (totalSteps - 1)) * 100;
            document.getElementById('progress-bar').style.width = `${progress}%`;
        }

        // Initial animation for step 1
        gsap.from("#step-1 .glass-card", { y: 40, opacity: 0, duration: 1.2, ease: "power3.out" });
        gsap.from("#step-1 .glass-card > *", { y: 20, opacity: 0, duration: 0.8, stagger: 0.15, delay: 0.4, ease: "power2.out" });

        window.nextStep = (step) => {
            const currentEl = document.getElementById(`step-${currentStep}`);
            const nextEl = document.getElementById(`step-${step}`);
            
            if (!nextEl) return;

            // Animate out current
            gsap.to(currentEl, {
                scale: 0.95,
                opacity: 0,
                duration: 0.5,
                ease: "power2.inOut",
                onComplete: () => {
                    currentEl.classList.remove('active');
                    currentEl.style.visibility = 'hidden';
                }
            });

            // Animate in next
            nextEl.classList.add('active');
            nextEl.style.visibility = 'visible';
            gsap.fromTo(nextEl, 
                { scale: 1.05, opacity: 0, y: 30 },
                { scale: 1, opacity: 1, y: 0, duration: 0.7, delay: 0.4, ease: "power3.out" }
            );

            // Animate children stagger
            gsap.fromTo(`#step-${step} .glass-card > *`,
                { y: 20, opacity: 0 },
                { y: 0, opacity: 1, duration: 0.6, stagger: 0.1, delay: 0.6, ease: "power2.out" }
            );

            currentStep = step;
            updateProgress();
        };

        // --- Polaroid 3D Hover Parallax ---
        const polaroidWrap = document.getElementById('polaroid-wrap');
        const polaroid = document.getElementById('polaroid-inner');
        
        if (polaroidWrap && polaroid) {
            polaroidWrap.addEventListener('mousemove', (e) => {
                const rect = polaroidWrap.getBoundingClientRect();
                const x = e.clientX - rect.left;
                const y = e.clientY - rect.top;
                
                const centerX = rect.width / 2;
                const centerY = rect.height / 2;
                
                const rotateX = ((y - centerY) / centerY) * -12;
                const rotateY = ((x - centerX) / centerX) * 12;
                
                polaroid.style.transform = `rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(1.05, 1.05, 1.05)`;
            });

            polaroidWrap.addEventListener('mouseleave', () => {
                polaroid.style.transform = 'rotate(-3deg)';
            });
        }

        // --- Finale Sequence ---
        window.celebrate = () => {
            const btn = document.getElementById('celebrate-btn');
            const wish = document.getElementById('final-wish');
            
            // 1. Disable & fade out button
            btn.disabled = true;
            gsap.to(btn, { opacity: 0, y: 20, scale: 0.9, duration: 0.5, ease: "power2.in", onComplete: () => btn.style.display = 'none' });
            
            // 2. Show final wish text sliding up
            wish.classList.remove('hidden');
            gsap.fromTo(wish, 
                { opacity: 0, y: 30 },
                { opacity: 1, y: 0, duration: 1.2, delay: 0.4, ease: "back.out(1.5)" }
            );
            
            // 3. Trigger Background 3D Hearts Finale
            if(window.triggerHeartsFinale) window.triggerHeartsFinale();

            // 4. Multi-layered Confetti Explosion
            const duration = 5 * 1000;
            const end = Date.now() + duration;

            // First: Powerful burst from bottom corners
            confetti({ particleCount: 150, spread: 80, origin: { y: 0.9, x: 0.1 }, colors: ['#ffc0cb', '#ff69b4', '#ffffff', '#ffd700'], startVelocity: 60 });
            confetti({ particleCount: 150, spread: 80, origin: { y: 0.9, x: 0.9 }, colors: ['#ffc0cb', '#ff69b4', '#ffffff', '#ffd700'], startVelocity: 60 });

            // Second: Continuous gentle shower of standard + emoji confetti
            (function frame() {
                // Left & Right continuous
                confetti({ particleCount: 4, angle: 60, spread: 55, origin: { x: 0, y: 0.6 }, colors: ['#ffc0cb', '#ff69b4', '#ff1493'] });
                confetti({ particleCount: 4, angle: 120, spread: 55, origin: { x: 1, y: 0.6 }, colors: ['#ffc0cb', '#ff69b4', '#ff1493'] });

                // Emoji Confetti Occasional Drops
                if (Math.random() < 0.25) {
                    const emojis = ['❤️', '💖', '✨', '🎂'];
                    confetti({
                        particleCount: 2,
                        scalar: 2 + Math.random(),
                        shapes: ['text'],
                        shapeOptions: {
                            text: { value: emojis[Math.floor(Math.random() * emojis.length)] }
                        },
                        origin: { x: Math.random(), y: -0.1 }, // Fall from very top
                        startVelocity: 0,
                        gravity: 0.8
                    });
                }

                if (Date.now() < end) {
                    requestAnimationFrame(frame);
                }
            }());
        };
    </script>
